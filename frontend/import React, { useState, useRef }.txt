const API = 'http://localhost:3001'; // em produção, URL da API

async function fetchSuggestions(ticketText, aoMudarEtapa) {
  aoMudarEtapa('Analisando o ticket…');

  const r = await fetch(`${API}/api/suggest`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ texto: ticketText }),
  });

  if (!r.ok) {
    throw new Error('Falha na análise do ticket.');
  }

  return await r.json();
}

const ETAPAS = [
  'Interpretando o ticket…',
  'Busca híbrida na base de conhecimento…',
  'Reordenando candidatos (rerank)…',
  'Gerando sugestão com citações…',
];

function tokenizar(texto) {
  return texto
    .toLowerCase()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .replace(/[^a-z0-9\- ]/g, ' ')
    .split(/\s+/)
    .filter((t) => t.length > 2);
}

function pontuar(ticketTokens, item) {
  const alvo = new Set(tokenizar(item.keywords + ' ' + item.titulo));
  let acertos = 0;
  for (const t of ticketTokens) if (alvo.has(t)) acertos += 1;
  const unicos = new Set(ticketTokens).size || 1;
  return Math.min(1, (acertos / unicos) * 1.9);
}

async function fetchSuggestions(ticketText, aoMudarEtapa) {
  for (let i = 0; i < ETAPAS.length; i++) {
    aoMudarEtapa(ETAPAS[i]);
    await new Promise((r) => setTimeout(r, 420 + Math.random() * 380));
  }

  const tokens = tokenizar(ticketText);
  const ranqueados = BASE_CONHECIMENTO.map((item) => ({
    ...item,
    score: pontuar(tokens, item),
  }))
    .sort((a, b) => b.score - a.score)
    .slice(0, 3)
    .filter((r) => r.score > 0.08);

  const melhor = ranqueados[0]?.score ?? 0;
  const confianca = melhor >= 0.62 ? 'alta' : melhor >= 0.34 ? 'media' : 'baixa';
  const abstencao = melhor < 0.16;

  const moduloPrincipal = ranqueados[0]?.modulo;
  const docs = moduloPrincipal ? DOCS_RELACIONADOS[moduloPrincipal] ?? [] : [];

  return {
    abstencao,
    confianca,
    similares: ranqueados,
    procedimento: abstencao
      ? []
      : [
          `Confirmar com o cliente o cenário descrito em ${ranqueados[0].id}: ${ranqueados[0].causa}`,
          ranqueados[0].solucao,
          'Validar o resultado com o cliente e registrar a causa raiz no encerramento do ticket.',
        ],
    docs,
  };
}

/* ---------- componentes ---------- */

const ROTULO_CONFIANCA = {
  alta: { texto: 'Confiança alta', classe: 'conf-alta' },
  media: { texto: 'Confiança média', classe: 'conf-media' },
  baixa: { texto: 'Confiança baixa', classe: 'conf-baixa' },
};

function BarraScore({ score }) {
  const pct = Math.round(score * 100);
  const nivel = score >= 0.62 ? 'alta' : score >= 0.34 ? 'media' : 'baixa';
  return (
    <div className="score-bloco" title={`Similaridade: ${pct}%`}>
      <span className={`score-num score-${nivel}`}>{pct}%</span>
      <div className="score-trilha">
        <div className={`score-preench score-bg-${nivel}`} style={{ width: `${pct}%` }} />
      </div>
    </div>
  );
}

function CartaoSimilar({ item, expandido, aoExpandir }) {
  return (
    <article className={`cartao ${expandido ? 'cartao-aberto' : ''}`}>
      <button className="cartao-cab" onClick={aoExpandir} aria-expanded={expandido}>
        <div className="cartao-meta">
          <span className="chip chip-id">{item.id}</span>
          <span className="chip">{item.modulo}</span>
          <span className="chip">{item.produto}</span>
        </div>
        <h3 className="cartao-titulo">{item.titulo}</h3>
        <BarraScore score={item.score} />
      </button>
      {expandido && (
        <div className="cartao-corpo">
          <p>
            <strong>Causa raiz:</strong> {item.causa}
          </p>
          <p>
            <strong>Solução aplicada:</strong> {item.solucao}
          </p>
          <p className="cartao-rodape">
            Resolvido por {item.resolvidoPor} em {item.data}
          </p>
        </div>
      )}
    </article>
  );
}

export default function App() {
  const [texto, setTexto] = useState('');
  const [etapa, setEtapa] = useState(null);
  const [resultado, setResultado] = useState(null);
  const [expandido, setExpandido] = useState(null);
  const [feedback, setFeedback] = useState(null);
  const [toast, setToast] = useState(null);
  const toastTimer = useRef(null);

  const analisando = etapa !== null;

  async function analisar() {
    if (!texto.trim() || analisando) return;
    setResultado(null);
    setFeedback(null);
    setExpandido(0);
    const r = await fetchSuggestions(texto, setEtapa);
    setEtapa(null);
    setResultado(r);
  }

  function registrarFeedback(valor) {
    setFeedback(valor);

    fetch(`${API}/api/feedback`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ ticket: texto, sugestao: resultado, resultado: valor }),
    }).catch((err) => console.error('Erro ao enviar feedback:', err));

    setToast('Feedback registrado — obrigado, ele realimenta a base.');
    clearTimeout(toastTimer.current);
    toastTimer.current = setTimeout(() => setToast(null), 3200);
  }

  return (
    <div className="app">
      <style>{CSS}</style>

      <header className="topo">
        <div className="marca">
          <span className="marca-logo">Ti</span>
          <div>
            <h1>TicketInsight</h1>
            <p>Triagem inteligente — SAP Business One · BR ONE</p>
          </div>
        </div>
        <span className="chip chip-poc">POC · base simulada</span>
      </header>

      <main className="grade">
        <section className="painel painel-entrada">
          <h2>Novo ticket</h2>
          <p className="dica">
            Cole a descrição do chamado. Em produção, este campo é preenchido automaticamente pelo
            webhook do sistema de chamados.
          </p>
          <textarea
            value={texto}
            onChange={(e) => setTexto(e.target.value)}
            placeholder="Descreva o problema relatado pelo cliente…"
            rows={8}
            disabled={analisando}
          />
          <div className="exemplos">
            <span>Exemplos:</span>
            {EXEMPLOS.map((ex) => (
              <button
                key={ex.rotulo}
                className="botao-exemplo"
                onClick={() => setTexto(ex.texto)}
                disabled={analisando}
              >
                {ex.rotulo}
              </button>
            ))}
          </div>
          <button className="botao-primario" onClick={analisar} disabled={!texto.trim() || analisando}>
            {analisando ? etapa : 'Analisar ticket'}
          </button>
          {analisando && <div className="barra-progresso" />}
        </section>

        <section className="painel painel-resultado">
          {!resultado && !analisando && (
            <div className="vazio">
              <p>As sugestões da IA aparecem aqui.</p>
              <p className="dica">
                O sistema busca chamados resolvidos, artigos e documentação do SAP B1 e do BR ONE
                semelhantes ao ticket informado.
              </p>
            </div>
          )}

          {analisando && (
            <div className="vazio">
              <p className="pulsando">{etapa}</p>
            </div>
          )}

          {resultado && resultado.abstencao && (
            <div className="abstencao">
              <h2>Nenhum caso suficientemente similar</h2>
              <p>
                A base não contém chamados próximos o bastante deste cenário. Trate como caso novo —
                ao resolvê-lo com boa documentação, ele passará a ajudar os próximos analistas.
              </p>
            </div>
          )}

          {resultado && !resultado.abstencao && (
            <>
              <div className="resultado-cab">
                <h2>Sugestão do TicketInsight</h2>
                <span className={`selo ${ROTULO_CONFIANCA[resultado.confianca].classe}`}>
                  {ROTULO_CONFIANCA[resultado.confianca].texto}
                </span>
              </div>

              <h3 className="secao">Procedimento recomendado</h3>
              <ol className="procedimento">
                {resultado.procedimento.map((p, i) => (
                  <li key={i}>{p}</li>
                ))}
              </ol>

              <h3 className="secao">
                Chamados similares <span className="contagem">{resultado.similares.length}</span>
              </h3>
              <div className="lista-cartoes">
                {resultado.similares.map((item, i) => (
                  <CartaoSimilar
                    key={item.id}
                    item={item}
                    expandido={expandido === i}
                    aoExpandir={() => setExpandido(expandido === i ? null : i)}
                  />
                ))}
              </div>

              {resultado.docs.length > 0 && (
                <>
                  <h3 className="secao">Documentação relacionada</h3>
                  <ul className="docs">
                    {resultado.docs.map((d) => (
                      <li key={d.ref}>
                        <span className="chip chip-id">{d.ref}</span> {d.titulo}
                      </li>
                    ))}
                  </ul>
                </>
              )}

              <div className="feedback">
                <p>Esta sugestão ajudou na resolução?</p>
                <div className="feedback-botoes">
                  {[
                    ['resolveu', 'Resolveu'],
                    ['parcial', 'Ajudou parcialmente'],
                    ['nao', 'Sem relação'],
                  ].map(([valor, rotulo]) => (
                    <button
                      key={valor}
                      className={`botao-feedback ${feedback === valor ? 'fb-ativo' : ''}`}
                      onClick={() => registrarFeedback(valor)}
                    >
                      {rotulo}
                    </button>
                  ))}
                </div>
              </div>
            </>
          )}
        </section>
      </main>

      {toast && <div className="toast">{toast}</div>}
    </div>
  );
}
